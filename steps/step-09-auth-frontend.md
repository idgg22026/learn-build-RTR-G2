# Step 9 — Login on the frontend

## Goal
The Vue app logs in, remembers who is logged in, sends people to the login page when the session ends, and blocks
pages a role may not open.

## You will learn
- One Axios instance with `withCredentials` + `withXSRFToken`
- Interceptors: retry on **419** (CSRF expired), redirect on **401**
- A Pinia store for the user and role helpers
- A router guard using `meta.roles`

## Do it

**`src/services/http.js`**
```js
import axios from 'axios'

const http = axios.create({
  baseURL: import.meta.env.VITE_API_URL,
  headers: { Accept: 'application/json' },
  withCredentials: true,   // send the login cookie
  withXSRFToken: true,     // send the XSRF token that protects it
})

const apiRoot = import.meta.env.VITE_API_URL.replace(/\/api\/?$/, '')

export function getCsrfCookie() {
  return axios.get(`${apiRoot}/sanctum/csrf-cookie`, { withCredentials: true })
}

http.interceptors.response.use(
  (response) => response,
  async (error) => {
    const status = error.response?.status
    const request = error.config

    // 419: the XSRF token expired. Get a new one and try once more.
    if (status === 419 && request && !request._retried) {
      request._retried = true
      await getCsrfCookie()
      return http(request)
    }

    // 401: the login has ended. Go to the login page, then come back.
    if (status === 401 && !request?.url?.endsWith('/me') && !request?.url?.endsWith('/login')) {
      const { default: router } = await import('@/router')
      const current = router.currentRoute.value
      if (current.name !== 'login') router.replace({ name: 'login', query: { redirect: current.fullPath } })
    }
    return Promise.reject(error)
  },
)

export default http
```

**`src/stores/auth.js`**
```js
import { defineStore } from 'pinia'
import http, { getCsrfCookie } from '@/services/http'

export const useAuthStore = defineStore('auth', {
  state: () => ({ user: null, checked: false }),
  getters: {
    role: (s) => s.user?.role ?? null,
    isOwner: (s) => s.user?.role === 'owner',
    canEdit: (s) => ['owner', 'staff'].includes(s.user?.role),
    isTenant: (s) => s.user?.role === 'tenant',
    home: (s) => (s.user?.role === 'tenant' ? '/portal' : '/'),
  },
  actions: {
    async fetchMe() {
      try { this.user = (await http.get('/me')).data.data } catch { this.user = null } finally { this.checked = true }
    },
    async login(phone, password) {
      await getCsrfCookie()
      const res = await http.post('/login', { phone, password })
      this.user = res.data.data
      this.checked = true
    },
    async logout() {
      try { await http.post('/logout') } finally { this.user = null }
    },
  },
})
```

**`src/router/index.js`**
```js
const STAFF = ['owner', 'staff', 'viewer']
const OWNER = ['owner']

const routes = [
  { path: '/login', name: 'login', component: () => import('@/pages/LoginPage.vue'), meta: { title: 'Log in', public: true } },
  {
    path: '/',
    component: AdminLayout,                        // step 10
    meta: { roles: STAFF },
    children: [
      { path: '', name: 'dashboard', component: () => import('@/pages/DashboardPage.vue'), meta: { title: 'Dashboard' } },
      { path: 'rooms', name: 'rooms', component: () => import('@/pages/rooms/RoomListPage.vue'), meta: { title: 'Rooms' } },
      { path: 'users', name: 'users', component: () => import('@/pages/users/UserListPage.vue'), meta: { title: 'Logins', roles: OWNER } },
    ],
  },
  { path: '/:pathMatch(.*)*', redirect: '/' },
]

const router = createRouter({ history: createWebHistory(import.meta.env.BASE_URL), routes })

router.beforeEach(async (to) => {
  const auth = useAuthStore()
  if (!auth.checked) await auth.fetchMe()

  if (to.meta.public) return auth.user ? auth.home : true
  if (!auth.user) return { name: 'login', query: to.fullPath === '/' ? {} : { redirect: to.fullPath } }

  // meta.roles of the page AND of its parent must all allow this role
  const allowed = to.matched.every((r) => !r.meta.roles || r.meta.roles.includes(auth.role))
  return allowed ? true : auth.home
})

router.afterEach((to) => { document.title = `${to.meta.title ?? 'Home'} | RTR Room` })
export default router
```

**`src/pages/LoginPage.vue`** (core)
```vue
<template>
  <div class="login-page">
    <div class="card login-card">
      <div class="card-body">
        <h3 class="text-center"><i class="fas fa-building text-primary"></i> RTR Room</h3>
        <form @submit.prevent="submit">
          <input v-model="phone" class="form-control mb-2" inputmode="tel" placeholder="012 345 678" autofocus />
          <input v-model="password" type="password" class="form-control mb-2" placeholder="Password" />
          <div v-if="error" class="alert alert-danger py-2">{{ error }}</div>
          <button class="btn btn-primary btn-block" :disabled="busy">
            <i class="fas fa-sign-in-alt mr-1"></i> Log in
          </button>
        </form>
      </div>
    </div>
  </div>
</template>

<script setup>
import { ref } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useAuthStore } from '@/stores/auth'

const auth = useAuthStore(), router = useRouter(), route = useRoute()
const phone = ref(''), password = ref(''), error = ref(''), busy = ref(false)

async function submit() {
  busy.value = true; error.value = ''
  try {
    await auth.login(phone.value, password.value)
    router.replace(route.query.redirect || auth.home)
  } catch (e) {
    error.value = e.response?.data?.errors?.phone?.[0] ?? e.response?.data?.message ?? 'Login failed.'
  } finally { busy.value = false }
}
</script>
```

## Practice
Add the global helper `$can` in `main.js` (used in every template to hide buttons):
```js
app.config.globalProperties.$can = (level) => {
  const auth = useAuthStore()
  return level === 'owner' ? auth.isOwner : auth.canEdit
}
```
Then use `<button v-if="$can('owner')">Delete</button>`. Remember: hiding is only for comfort. **The server checks too.**
